# Use this command to run the rabbitmq docker image

sudo docker run -d --rm --name rabbitmq -p 5672:5672 -p 15672:15672 rabbitmq:4-management

# the deb version can be started with:

sudo systemctl start rabbitmq-server

or

sudo rabbitmq-server -detached

# Then login on the browser to:

http://localhost:15672

# Default password and login are "guest" for both 

==

# For the deb version of rabbitmq, you must enable the management interface first after install:

sudo rabbitmq-plugins enable rabbitmq_management

# Then restart the systemd service:

sudo systemctl restart rabbitmq-server

# then check if the management console is listening 

sudo rabbitmq-diagnostics listeners

# Look for a line like "Interface: [::], port: 15672, protocol: http, purpose: HTTP API"
# And if this is part of the output, the managent console has been enabled. 

==

# To deactivate RabbitMQ and prevent it from starting automatically when Linux Mint boots, 
# run the standard systemd service management commands: 

sudo systemctl stop rabbitmq-server 

and 

sudo systemctl disable rabbitmq-server

# Whenever you sit down to study your RabbitMQ tutorials later, spin up the node manually with a single command:

sudo systemctl start rabbitmq-server

# Once your study session is finished, you can close it down again using the stop command.
